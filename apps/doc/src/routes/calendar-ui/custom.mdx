# 第三方适配器

不使用 `@sanguo-dev/calendar`，自己实现一个 `CalendarAdapter`；示例数据、取数方法和适配器都直接写在演示文件里。

<CalendarUICustom />

```ts
// 第三方数据源：按年列出放假 / 补班的 MMDD，names 提供副标题
const COMPANY_REST: Record<
  string,
  { xiu?: string[]; ban?: string[]; names?: Record<string, string> }
> = {
  2026: {
    xiu: ["0101", "0501", "0605", "1001"],
    ban: ["0613"],
    names: {
      "0101": "公司元旦",
      "0501": "公司劳动节",
      "0605": "六月公司日",
      "0613": "公司日补班",
      "1001": "公司国庆",
    },
  },
};

const pad = (value: number) => String(value).padStart(2, "0");

interface CompanyRest {
  date: string;
  type: "holiday" | "makeup";
  subtitle: string;
}

/** 第三方接口：按日期返回节假日 / 调休数据 */
function getCompanyRest(date: Date): CompanyRest | null {
  const year = COMPANY_REST[String(date.getFullYear())];
  if (!year) return null;

  const mmdd = `${pad(date.getMonth() + 1)}${pad(date.getDate())}`;
  const day = `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
  const subtitle = year.names?.[mmdd];

  if (year.ban?.includes(mmdd)) {
    return { date: day, type: "makeup", subtitle: subtitle ?? "公司补班" };
  }
  if (year.xiu?.includes(mmdd)) {
    return { date: day, type: "holiday", subtitle: subtitle ?? "公司假日" };
  }
  return null;
}

const adapter: CalendarAdapter<CompanyRest> = {
  id: "company-holidays",
  decorateMonth({ days }) {
    return days.map((date) => {
      const rest = getCompanyRest(date.date);
      if (!rest) return null;

      return {
        subtitle: rest.subtitle,
        badge:
          rest.type === "holiday" ? { text: "休", tone: "rest" } : { text: "班", tone: "work" },
        data: rest,
      };
    });
  },
};

createCalendarUI(container, { adapter });
```

适配器接口、`DayDecoration` 等字段见 [CalendarAdapter](/api/calendar-ui/types/calendar-adapter)。
